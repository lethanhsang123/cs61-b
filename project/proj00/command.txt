- Change working directory to prj0
- create libs folder:
    + algs4.jar
    + junit-platform-console-standalone-6.1.2.jar

Step 1: Compile Time
    - Compile production code: javac -cp "libs/algs4.jar" -d out/classes $(find src/main/java -name "*.java")
    - Compile tests: javac -cp "out/classes:libs/algs4.jar:libs/junit-platform-console-standalone-6.1.2.jar" -d out/test-classes $(find src/test/java -name "*.java")

Step 2: Run
    - Mode 1: 
        + Run application (out/classes + libs): java -cp "out/classes:libs/algs4.jar" edu.berkeley.cs61b.proj0.Main
        + Run JUnit tests (out/classes + out/test-classes + libs): java -jar libs/junit-platform-console-standalone-6.1.2.jar execute --class-path "out/classes:out/test-classes:libs/algs4.jar" --scan-class-path
    - Mode 2:
        + JAR Packagig: jar --create --file dist/proj0.jar --main-class edu.berkeley.cs61b.proj0.Main -C out/classes .
        + Run JUnit tests (jar + out/test-classes + libs): java -jar libs/junit-platform-console-standalone-6.1.2.jar execute --class-path "dist/proj0.jar:out/test-classes:libs/algs4.jar" --scan-class-path
        + Run application (jar + libs): java -cp "dist/proj0.jar:libs/algs4.jar" edu.berkeley.cs61b.proj0.Main
